local FLY_ENABLED = false

    InputType = Enum.UserInputType.Keyboard,
    KeyCode = Enum.KeyCode.F
}

local flyKeys = {W=false, A=false, S=false, D=false, Up=false, Down=false}
local flyAttachment = nil
local flyVelocity = nil
local flyOrientation = nil

local function stopFly()
    if flyVelocity then flyVelocity:Destroy(); flyVelocity = nil end
    if flyOrientation then flyOrientation:Destroy(); flyOrientation = nil end
    if flyAttachment then flyAttachment:Destroy(); flyAttachment = nil end
    local c = LP.Character
    local h = c and c:FindFirstChildOfClass("Humanoid")
    if h then h.AutoRotate = true end
end

local function startFly()
    stopFly()
    local c = LP.Character
    local root = c and c:FindFirstChild("HumanoidRootPart")
    local h = c and c:FindFirstChildOfClass("Humanoid")
    if not root or not h then return end
    flyAttachment = Instance.new("Attachment")
    flyAttachment.Name = "ZalanFlyAttachment"
    flyAttachment.Parent = root
    flyVelocity = Instance.new("LinearVelocity")
    flyVelocity.Name = "ZalanFlyVelocity"
    flyVelocity.Attachment0 = flyAttachment
    flyVelocity.RelativeTo = Enum.ActuatorRelativeTo.World
    flyVelocity.VelocityConstraintMode = Enum.VelocityConstraintMode.Vector
    flyVelocity.MaxForce = math.huge
    flyVelocity.VectorVelocity = Vector3.zero
    flyVelocity.Parent = root
    flyOrientation = Instance.new("AlignOrientation")
    flyOrientation.Name = "ZalanFlyOrientation"
    flyOrientation.Attachment0 = flyAttachment
    flyOrientation.Mode = Enum.OrientationAlignmentMode.OneAttachment
    flyOrientation.RigidityEnabled = false
    flyOrientation.Responsiveness = 25
    flyOrientation.MaxTorque = math.huge
    flyOrientation.Parent = root
    h.AutoRotate = false
end

local function setFly(enabled)
    FLY_ENABLED = enabled
    if enabled then startFly() else stopFly() end
end

local function setSuperJump(enabled)
    SUPER_JUMP_ENABLED = enabled
    local c = LP.Character
    local h = c and c:FindFirstChildOfClass("Humanoid")
    if h then
        h.UseJumpPower = true
        h.JumpPower = enabled and SUPER_JUMP_POWER or 50
    end
end

local function setNoclip(enabled)
    NOCLIP_ENABLED = enabled
    if not enabled then
        for part, value in pairs(noclipOriginal) do
            if part and part.Parent then part.CanCollide = value end
        end
        table.clear(noclipOriginal)
    end
end

local function setRainbow(enabled)
    RAINBOW_ENABLED = enabled
    local c = LP.Character
    if not c then return end
    if not enabled then
        for part, color in pairs(rainbowOriginal) do
            if part and part.Parent then part.Color = color end
        end
        table.clear(rainbowOriginal)
    end
end

local function resetExtraTrolls()
    setSuperJump(false)
    setNoclip(false)
    setRainbow(false)
    DRUNK_CAMERA_ENABLED = false
    FOV_SPAM_ENABLED = false
    UPSIDE_CAMERA_ENABLED = false
    Camera.FieldOfView = ORIGINAL_FOV
end

local function applyLocalInvisible(enabled)
    INVISIBLE_ENABLED = enabled
    local character = LP.Character
    if not character then return end
    for _, obj in ipairs(character:GetDescendants()) do
        if obj:IsA("BasePart") then
            obj.LocalTransparencyModifier = enabled and 1 or 0
        end
    end
end

local function setMoon(enabled)
    MOON_ENABLED = enabled
    workspace.Gravity = enabled and 70 or ORIGINAL_GRAVITY
end

--==============================================================
-- FLY RENDER LOOP
--==============================================================

connect(RunService.RenderStepped, function()
    if UNLOADED or not FLY_ENABLED then return end
    local c = LP.Character
    local root = c and c:FindFirstChild("HumanoidRootPart")
    if not root then return end
    if not flyVelocity or not flyVelocity.Parent then startFly() end
    if not flyVelocity then return end
    local look = Camera.CFrame.LookVector
    local right = Camera.CFrame.RightVector
    local forward = Vector3.new(look.X,0,look.Z)
    local side = Vector3.new(right.X,0,right.Z)
    if forward.Magnitude > 0 then forward = forward.Unit end
    if side.Magnitude > 0 then side = side.Unit end
    local dir = Vector3.zero
    if flyKeys.W then dir += forward end
    if flyKeys.S then dir -= forward end
    if flyKeys.D then dir += side end
    if flyKeys.A then dir -= side end
    if flyKeys.Up then dir += Vector3.yAxis end
    if flyKeys.Down then dir -= Vector3.yAxis end
    if dir.Magnitude > 1 then dir = dir.Unit end
    flyVelocity.VectorVelocity = dir * FLY_SPEED
    if flyOrientation and forward.Magnitude > 0 then
        flyOrientation.CFrame = CFrame.lookAt(Vector3.zero, forward)
    end
end)

local waitingBind = nil


local function bindName(bind)

    if bind.InputType
        ==
        Enum.UserInputType.Keyboard
    then

        return bind.KeyCode.Name

    end


    if bind.InputType
        ==
        Enum.UserInputType.MouseButton1
    then

        return "MOUSE 1"

    end


    if bind.InputType
        ==
        Enum.UserInputType.MouseButton2
    then

        return "MOUSE 2"

    end


    if bind.InputType
        ==
        Enum.UserInputType.MouseButton3
    then

        return "MOUSE 3"

    end


    return bind.InputType.Name

end


local function validBindInput(input)

    if input.UserInputType
        ==
        Enum.UserInputType.Keyboard
    then

        return
            input.KeyCode
            ~=
            Enum.KeyCode.Unknown

    end


    return

        input.UserInputType
        ==
        Enum.UserInputType.MouseButton1

        or

        input.UserInputType
        ==
        Enum.UserInputType.MouseButton2

        or

        input.UserInputType
        ==
        Enum.UserInputType.MouseButton3

end


local function inputToBind(input)

    if input.UserInputType
        ==
        Enum.UserInputType.Keyboard
    then

        return {

            InputType =
                Enum.UserInputType.Keyboard,

            KeyCode =
                input.KeyCode
        }

    end


    return {

        InputType =
            input.UserInputType,

        KeyCode =
            nil
    }

end


local function matchesBind(
    input,
    bind
)

    if bind.InputType
        ==
        Enum.UserInputType.Keyboard
    then

        return

            input.UserInputType
            ==
            Enum.UserInputType.Keyboard

            and

            input.KeyCode
            ==
            bind.KeyCode

    end


    return

        input.UserInputType
        ==
        bind.InputType

end

local function newLine()

    local line =
        Instance.new(
            "Frame"
        )

    line.AnchorPoint =
        Vector2.new(
            0.5,
            0.5
        )

    line.BackgroundColor3 =
        line:GetAttribute("ESPLineColor") or ESP_COLOR

    line.BorderSizePixel = 0

    line.Visible = false

    line.ZIndex = 20

    line.Parent =
        Overlay

    return line

end


local function setLine(
    line,
    a,
    b,
    thickness
)

    local delta =
        b - a

    if delta.Magnitude <= 0 then

        line.Visible =
            false

        return

    end


    line.Size =
        UDim2.fromOffset(
            delta.Magnitude,
            thickness
        )


    line.Position =
        UDim2.fromOffset(
            (
                a.X
                +
                b.X
            )
            /
            2,

            (
                a.Y
                +
                b.Y
            )
            /
            2
        )


    line.Rotation =
        math.deg(
            math.atan2(
                delta.Y,
                delta.X
            )
        )


    line.BackgroundColor3 =
        ESP_COLOR

    line.Visible =
        true

end


local function hideLines(lines)

    for _, line in ipairs(lines) do

        line.Visible =
            false

    end

end

local function updateBox(
    data,
    character
)

    for _, line in ipairs(data.Box) do
        line:SetAttribute("ESPLineColor", ESP_COLORS.Box)
    end

    if not ESP_BOX then

        hideLines(
            data.Box
        )

        return

    end


    local root =
        character:
        FindFirstChild(
            "HumanoidRootPart"
        )

    local head =
        character:
        FindFirstChild(
            "Head"
        )


    if not root
        or not head
    then

        hideLines(
            data.Box
        )

        return

    end


    local rootScreen,
          rootVisible =

        Camera:
        WorldToViewportPoint(
            root.Position
        )


    local topScreen,
          topVisible =

        Camera:
        WorldToViewportPoint(
            head.Position
            +
            Vector3.new(
                0,
                0.8,
                0
            )
        )


    local bottomScreen,
          bottomVisible =

        Camera:
        WorldToViewportPoint(
            root.Position
            -
            Vector3.new(
                0,
                3,
                0
            )
        )


    if not rootVisible
        or not topVisible
        or not bottomVisible
    then

        hideLines(
            data.Box
        )

        return

    end


    local height =
        math.abs(
            bottomScreen.Y
            -
            topScreen.Y
        )


    if height < 5 then

        hideLines(
            data.Box
        )

        return

    end


    local width =
        height * 0.55


    local x =
        rootScreen.X


    local TL =
        Vector2.new(
            x - width / 2,
            topScreen.Y
        )

    local TR =
        Vector2.new(
            x + width / 2,
            topScreen.Y
        )

    local BL =
        Vector2.new(
            x - width / 2,
            bottomScreen.Y
        )

    local BR =
        Vector2.new(
            x + width / 2,
            bottomScreen.Y
        )


    setLine(
        data.Box[1],
        TL,
        TR,
        2
    )

    setLine(
        data.Box[2],
        TR,
        BR,
        2
    )

    setLine(
        data.Box[3],
        BR,
        BL,
        2
    )

    setLine(
        data.Box[4],
        BL,
        TL,
        2
    )

end

local function updateSkeleton(
    data,
    character,
    humanoid
)

    for _, line in ipairs(data.Skeleton) do
        line:SetAttribute("ESPLineColor", ESP_COLORS.Skeleton)
    end

    if not ESP_SKELETON then

        hideLines(
            data.Skeleton
        )

        return

    end


    hideLines(
        data.Skeleton
    )


    local definition =

        humanoid.RigType
        ==
        Enum.HumanoidRigType.R15

        and R15
        or R6


    for index, pair in ipairs(
        definition
    ) do

        local a =
            character:
            FindFirstChild(
                pair[1]
            )

        local b =
            character:
            FindFirstChild(
                pair[2]
            )

        local line =
            data.Skeleton[index]


        if a
            and b
            and line
        then

            local sa,
                  va =

                Camera:
                WorldToViewportPoint(
                    a.Position
                )


            local sb,
                  vb =

                Camera:
                WorldToViewportPoint(
                    b.Position
                )


            if va
                and vb
                and sa.Z > 0
                and sb.Z > 0
            then

                setLine(
                    line,

                    Vector2.new(
                        sa.X,
                        sa.Y
                    ),

                    Vector2.new(
                        sb.X,
                        sb.Y
                    ),

                    2
                )

            end

        end

    end

end

local function updateHitbox(player)

    if player == LP
        or isWhitelisted(player)
    then

        removeHitbox(
            player
        )

        return

    end


    if not HITBOX_ENABLED then

        removeHitbox(
            player
        )

        return

    end


    local character =
        player.Character


    if not character then

        removeHitbox(
            player
        )

        return

    end


    local humanoid =
        character:
        FindFirstChildOfClass(
            "Humanoid"
        )


    local target =
        getHitboxTarget(
            character
        )


    if not humanoid
        or humanoid.Health <= 0
        or not target
    then

        removeHitbox(
            player
        )

        return

    end


    local hitbox =
        Hitboxes[player]


    if hitbox then

        local targetValue =
            hitbox:
            FindFirstChild(
                "TargetPart"
            )

        if not targetValue
            or targetValue.Value
            ~= target
        then

            removeHitbox(
                player
            )

            hitbox =
                nil

        end

    end


    if not hitbox then

        hitbox =
            Instance.new(
                "Part"
            )

        hitbox.Name =
            "ZalanDebugHitbox"

        hitbox.Anchored =
            false

        hitbox.CanCollide =
            false

        hitbox.CanTouch =
            false

        hitbox.CanQuery =
            false

        hitbox.Massless =
            true

        hitbox.CastShadow =
            false

        hitbox.Material =
            Enum.Material.ForceField

        hitbox.CFrame =
            target.CFrame

        hitbox.Parent =
            character


        local targetValue =
            Instance.new(
                "ObjectValue"
            )

        targetValue.Name =
            "TargetPart"

        targetValue.Value =
            target

        targetValue.Parent =
            hitbox


        local weld =
            Instance.new(
                "WeldConstraint"
            )

        weld.Part0 =
            target

        weld.Part1 =
            hitbox

        weld.Parent =
            hitbox


        Hitboxes[player] =
            hitbox

    end


    hitbox.Size =
        target.Size
        *
        HITBOX_SCALE


    hitbox.Color =
        ESP_COLOR


    hitbox.Transparency =

        HITBOX_VISIBLE

        and 0.65

        or 1

end

    if enabled
        and ESP[player]
    then

        hideESP(
            ESP[player]
        )

    end


    if enabled then

        removeHitbox(
            player
        )

    end

end

GUI:SetAttribute("TeamCheck", false)
GUI:SetAttribute("WallCheck", false)


local function validAimPlayer(
    player
)

    if not player
        or player == LP
    then

        return false

    end


    if isWhitelisted(
        player
    ) then

        return false

    end


    if GUI:GetAttribute("TeamCheck")
        and LP.Team ~= nil
        and player.Team ~= nil
        and LP.Team == player.Team
    then

        return false

    end


    local character =
        player.Character


    if not character then

        return false

    end


    local humanoid =
        character:
        FindFirstChildOfClass(
            "Humanoid"
        )


    local part =
        getAimPart(
            character
        )


    if not humanoid
        or humanoid.Health <= 0
        or not part
    then

        return false

    end


    local distance =
        (
            part.Position
            -
            Camera.CFrame.Position
        ).Magnitude


    if distance
        >
        AIM_DISTANCE
    then

        return false

    end


    if GUI:GetAttribute("WallCheck") then

        local rayParams =
            RaycastParams.new()

        rayParams.FilterType =
            Enum.RaycastFilterType.Exclude

        if LP.Character then
            rayParams.FilterDescendantsInstances =
                {LP.Character}
        else
            rayParams.FilterDescendantsInstances =
                {}
        end

        rayParams.IgnoreWater =
            true

        local wallResult =
            workspace:Raycast(
                Camera.CFrame.Position,
                part.Position - Camera.CFrame.Position,
                rayParams
            )

        if wallResult
            and not wallResult.Instance:IsDescendantOf(
                character
            )
        then

            return false

        end

    end


    local screen,
          visible =

        Camera:
        WorldToViewportPoint(
            part.Position
        )


    if not visible
        or screen.Z <= 0
    then

        return false

    end


    local center =
        Vector2.new(
            Camera.ViewportSize.X
            /
            2,

            Camera.ViewportSize.Y
            /
            2
        )


    local screenDistance =
        (
            Vector2.new(
                screen.X,
                screen.Y
            )
            -
            center
        ).Magnitude


    return

        screenDistance
        <=
        AIM_FOV

end


--==============================================================
-- FIND TARGET
--==============================================================

local function findAimPlayer()

    local closest =
        nil

    local closestDistance =
        math.huge


    local center =
        Vector2.new(
            Camera.ViewportSize.X
            /
            2,

            Camera.ViewportSize.Y
            /
            2
        )


    for _, player in ipairs(
        Players:GetPlayers()
    ) do

        if validAimPlayer(
            player
        ) then

            local part =
                getAimPart(
                    player.Character
                )


            local screen =
                Camera:
                WorldToViewportPoint(
                    part.Position
                )


            local distance =
                (
                    Vector2.new(
                        screen.X,
                        screen.Y
                    )
                    -
                    center
                ).Magnitude


            if distance
                <
                closestDistance
            then

                closestDistance =
                    distance

                closest =
                    player

            end

        end

    end


    return closest

end


--==============================================================
-- SILENT AIM SHOT DIRECTION
--==============================================================

local function getShotDirection(origin, normalDirection)
    if not SILENT_AIM_ENABLED then
        return normalDirection
    end

    local target = findAimPlayer()

    if not target or not validAimPlayer(target) then
        return normalDirection
    end

    local targetPart = getAimPart(target.Character)

    if not targetPart then
        return normalDirection
    end

    local delta = targetPart.Position - origin

    if delta.Magnitude <= 0 then
        return normalDirection
    end

    return delta.Unit
end

-- LOOK TELEPORT
--==============================================================

do
    local TELEPORT_BIND = {
        InputType = Enum.UserInputType.Keyboard,
        KeyCode = Enum.KeyCode.T
    }

    local function getLookedAtPlayer()
        local viewport = Camera.ViewportSize
        local center = Vector2.new(viewport.X / 2, viewport.Y / 2)
        local ray = Camera:ViewportPointToRay(center.X, center.Y)

        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.IgnoreWater = true
        params.FilterDescendantsInstances = LP.Character and {LP.Character} or {}

        local result = workspace:Raycast(ray.Origin, ray.Direction * 5000, params)

        if result and result.Instance then
            local model = result.Instance:FindFirstAncestorOfClass("Model")
            local player = model and Players:GetPlayerFromCharacter(model)
            if player and player ~= LP then
                return player
            end
        end

        local bestPlayer = nil
        local bestPixels = 70

        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LP then
                local character = player.Character
                local humanoid = character and character:FindFirstChildOfClass("Humanoid")
                local root = character and character:FindFirstChild("HumanoidRootPart")

                if humanoid and humanoid.Health > 0 and root then
                    local pos, visible = Camera:WorldToViewportPoint(root.Position)
                    if visible and pos.Z > 0 then
                        local pixels = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                        if pixels < bestPixels then
                            bestPixels = pixels
                            bestPlayer = player
                        end
                    end
                end
            end
        end

        return bestPlayer
    end

    local function teleportToLookedPlayer()
        local target = getLookedAtPlayer()
        if not target then
            if notify then notify("Look at a player first") end
            return
        end

        local myRoot = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        local targetRoot = target.Character and target.Character:FindFirstChild("HumanoidRootPart")

        if not myRoot or not targetRoot then
            if notify then notify("Teleport unavailable") end
            return
        end

        myRoot.CFrame = targetRoot.CFrame * CFrame.new(3, 0, 0)
        if notify then notify("Teleported to " .. target.Name) end
    end

    local card = createCard(PlayerPage, 64)
    local label = makeText(card, "LOOK TELEPORT", 13, TEXT, Enum.Font.GothamMedium)
    label.Size = UDim2.new(1, -190, 1, 0)
    label.Position = UDim2.fromOffset(18, 0)

    local button = Instance.new("TextButton")
    button.Size = UDim2.fromOffset(140, 34)
    button.Position = UDim2.new(1, -158, 0.5, -17)
    button.BackgroundColor3 = CARD2
    button.BorderSizePixel = 0
    button.Text = "TELEPORT"
    button.TextColor3 = ACCENT
    button.TextSize = 11
    button.Font = Enum.Font.GothamBold
    button.AutoButtonColor = false
    button.Parent = card
    corner(button, 7)
    stroke(button)

    connect(button.MouseButton1Click, teleportToLookedPlayer)

    createKeybind(PlayerPage, "TELEPORT BIND", TELEPORT_BIND, function(newBind)
        TELEPORT_BIND = newBind
    end)

    connect(UIS.InputBegan, function(input, gameProcessed)
        if gameProcessed or UNLOADED then return end
        if waitingBind then return end
        if matchesBind(input, TELEPORT_BIND) then
            teleportToLookedPlayer()
        end
    end)
end

    if now - fpsClock >= 0.5 then
        local elapsed = now - fpsClock
        local shownFPS = elapsed > 0 and math.floor(fpsFrames / elapsed + 0.5) or 0
        fpsFrames = 0
        fpsClock = now
        local session = math.max(0, math.floor(now - SESSION_STARTED))

-- ESP + HITBOX RENDER
--==============================================================

connect(
    RunService.RenderStepped,
    function()

        if UNLOADED then
            return
        end


        local myCharacter =
            LP.Character


        local myRoot =
            myCharacter
            and
            myCharacter:
            FindFirstChild(
                "HumanoidRootPart"
            )


        for _, player in ipairs(
            Players:GetPlayers()
        ) do

            if player == LP then
                continue
            end


            if not ESP[player] then

                createESP(
                    player
                )

            end


            local data =
                ESP[player]


            --==================================================
            -- WHITELIST
            --==================================================

            if isWhitelisted(
                player
            )
                or (
                    GUI:GetAttribute("TeamCheck")
                    and LP.Team ~= nil
                    and player.Team ~= nil
                    and LP.Team == player.Team
                )
            then

                hideESP(
                    data
                )

                removeHitbox(
                    player
                )

                continue

            end


            --==================================================
            -- DEBUG HITBOX
            --==================================================

            if HITBOX_ENABLED then

                updateHitbox(
                    player
                )

            else

                removeHitbox(
                    player
                )

            end


            --==================================================
            -- ESP OFF
            --==================================================

            if not ESP_ENABLED
                or not myRoot
            then

                hideESP(
                    data
                )

                continue

            end


            local character =
                player.Character


            local humanoid =
                character
                and
                character:
                FindFirstChildOfClass(
                    "Humanoid"
                )


            local root =
                character
                and
                character:
                FindFirstChild(
                    "HumanoidRootPart"
                )


            local head =
                character
                and
                character:
                FindFirstChild(
                    "Head"
                )


            if not character
                or not humanoid
                or not root
                or not head
                or humanoid.Health <= 0
            then

                hideESP(
                    data
                )

                continue

            end


            local distance =
                (
                    myRoot.Position
                    -
                    root.Position
                ).Magnitude


            if distance
                >
                ESP_MAX_DISTANCE
            then

                hideESP(
                    data
                )

                continue

            end


            local screen,
                  visible =

                Camera:
                WorldToViewportPoint(
                    head.Position
                    +
                    Vector3.new(
                        0,
                        0.8,
                        0
                    )
                )


            if not visible
                or screen.Z <= 0
            then

                hideESP(
                    data
                )

                continue

            end


            --==================================================
            -- HIGHLIGHT
            --==================================================

            data.Highlight.Adornee =
                character

            data.Highlight.FillColor =
                ESP_COLORS.Highlight

            data.Highlight.OutlineColor =
                ESP_COLORS.Highlight

            data.Highlight.Enabled =
                ESP_HIGHLIGHT


            --==================================================
            -- INFO TEXT
            --==================================================

            local info = {}


            if ESP_NAME then

                table.insert(
                    info,
                    player.DisplayName
                )

            end


            if ESP_DISTANCE then

                table.insert(
                    info,
                    "Távolság: "
                    ..
                    math.floor(
                        distance
                    )
                )

            end


            if ESP_HEALTH then

                table.insert(
                    info,
                    "HP: "
                    ..
                    math.floor(
                        humanoid.Health
                    )
                    ..
                    "/"
                    ..
                    math.floor(
                        humanoid.MaxHealth
                    )
                )

            end


            data.Info.Text =
                table.concat(
                    info,
                    "\n"
                )


            data.Info.TextColor3 =
                ESP_COLOR


            data.Info.Position =
                UDim2.fromOffset(
                    screen.X,
                    screen.Y
                )


            data.Info.Visible =
                #info > 0


            updateBox(
                data,
                character
            )


            updateSkeleton(
                data,
                character,
                humanoid
            )

        end

    end
)

                Lerp(

                    strength
                )

-- INPUT BEGAN
--==============================================================

connect(
    UIS.InputBegan,
    function(
        input,
        processed
    )

        if UNLOADED then
            return
        end


        --======================================================
        -- KEYBIND CAPTURE
        --======================================================

        if waitingBind then

            if validBindInput(
                input
            ) then

                local callback =
                    waitingBind

                waitingBind =
                    nil

                callback(
                    input
                )

            end

            return

        end


        if processed then
            return
        end


        --======================================================
        -- INSERT
        --======================================================

        if input.UserInputType
            ==
            Enum.UserInputType.Keyboard

            and

            input.KeyCode
            ==
            GUI_KEY
        then

            Main.Visible =
                not Main.Visible

            return

        end


        --======================================================
        -- AIM
        --======================================================

        if matchesBind(
            input,
            AIM_BIND
        ) then

            if AIM_MODE
                ==
                "HOLD"
            then

                AIM_ENABLED =
                    true

            else

                AIM_ENABLED =
                    not AIM_ENABLED


                if not AIM_ENABLED then

                    currentTarget =
                        nil

                end

            end


            return

        end


        --======================================================
        -- FLY MOVEMENT / BIND
        --======================================================

        if input.KeyCode == Enum.KeyCode.W then flyKeys.W = true end
        if input.KeyCode == Enum.KeyCode.A then flyKeys.A = true end
        if input.KeyCode == Enum.KeyCode.S then flyKeys.S = true end
        if input.KeyCode == Enum.KeyCode.D then flyKeys.D = true end
        if input.KeyCode == Enum.KeyCode.Space then flyKeys.Up = true end
        if input.KeyCode == Enum.KeyCode.LeftControl or input.KeyCode == Enum.KeyCode.RightControl then flyKeys.Down = true end

        if matchesBind(input, FLY_BIND) then
            FLY_ENABLED = not FLY_ENABLED
            FlyToggle.Set(FLY_ENABLED)
            return
        end

        --======================================================
        -- SPEED
        --======================================================

        if matchesBind(
            input,
            SPEED_BIND
        ) then

            SPEED_ENABLED =
                not SPEED_ENABLED


            SpeedToggle.Set(
                SPEED_ENABLED
            )


            return

        end

    end
)

local function floatingBindName(bind)
    if bind == nil then
        return "-"
    end
    local ok, result = pcall(function()
        return bindName(bind)
    end)
    if ok and result ~= nil then
        return tostring(result)
    end
    return "-"
end
