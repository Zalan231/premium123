local function createSlider(
    parent,
    name,
    minimum,
    maximum,
    default,
    suffix,
    callback
)

    local card =
        createCard(
            parent,
            82
        )


    local title =
        makeText(
            card,
            name,
            13,
            TEXT,
            Enum.Font.GothamMedium
        )

    title.Size =
        UDim2.new(
            1,
            -120,
            0,
            28
        )

    title.Position =
        UDim2.fromOffset(
            18,
            8
        )


    local valueText =
        makeText(
            card,
            "",
            12,
            ACCENT,
            Enum.Font.GothamBold
        )

    valueText.Size =
        UDim2.fromOffset(
            100,
            28
        )

    valueText.Position =
        UDim2.new(
            1,
            -118,
            0,
            8
        )

    valueText.TextXAlignment =
        Enum.TextXAlignment.Right


    local slider =
        Instance.new(
            "Frame"
        )

    slider.Size =
        UDim2.new(
            1,
            -36,
            0,
            7
        )

    slider.Position =
        UDim2.fromOffset(
            18,
            53
        )

    slider.BackgroundColor3 =
        Color3.fromRGB(
            45,
            49,
            62
        )

    slider.BorderSizePixel = 0

    slider.Active = true

    slider.Parent =
        card

    corner(
        slider,
        4
    )


    local fill =
        Instance.new(
            "Frame"
        )

    fill.BackgroundColor3 =
        ACCENT

    fill.BorderSizePixel = 0

    fill.Parent =
        slider

    corner(
        fill,
        4
    )


    local knob =
        Instance.new(
            "Frame"
        )

    knob.Size =
        UDim2.fromOffset(
            15,
            15
        )

    knob.AnchorPoint =
        Vector2.new(
            0.5,
            0.5
        )

    knob.BackgroundColor3 =
        Color3.new(
            1,
            1,
            1
        )

    knob.BorderSizePixel = 0

    knob.Active = true

    knob.Parent =
        slider

    corner(
        knob,
        8
    )


    local value =
        default

    local dragging =
        false


    local function setValue(v)

        value =
            math.clamp(
                math.floor(v),
                minimum,
                maximum
            )

        local percent =
            (
                value
                -
                minimum
            )
            /
            (
                maximum
                -
                minimum
            )

        fill.Size =
            UDim2.fromScale(
                percent,
                1
            )

        knob.Position =
            UDim2.new(
                percent,
                0,
                0.5,
                0
            )

        valueText.Text =
            tostring(value)
            ..
            (
                suffix
                or ""
            )

        if callback then

            callback(
                value
            )

        end

    end


    local function fromMouse(x)

        if slider.AbsoluteSize.X <= 0 then
            return
        end

        local percent =
            math.clamp(
                (
                    x
                    -
                    slider.AbsolutePosition.X
                )
                /
                slider.AbsoluteSize.X,
                0,
                1
            )

        setValue(
            minimum
            +
            (
                maximum
                -
                minimum
            )
            *
            percent
        )

    end


    connect(
        slider.InputBegan,
        function(input)

            if input.UserInputType
                ==
                Enum.UserInputType.MouseButton1
            then

                dragging =
                    true

                fromMouse(
                    input.Position.X
                )

            end

        end
    )


    connect(
        knob.InputBegan,
        function(input)

            if input.UserInputType
                ==
                Enum.UserInputType.MouseButton1
            then

                dragging =
                    true

            end

        end
    )


    connect(
        UIS.InputChanged,
        function(input)

            if dragging
                and
                input.UserInputType
                ==
                Enum.UserInputType.MouseMovement
            then

                fromMouse(
                    input.Position.X
                )

            end

        end
    )


    connect(
        UIS.InputEnded,
        function(input)

            if input.UserInputType
                ==
                Enum.UserInputType.MouseButton1
            then

                dragging =
                    false

            end

        end
    )


    setValue(
        default
    )


    return {

        Get =
            function()

                return value

            end,

        Set =
            setValue

    }

end

local waitingBind = nil


local function bindName(bind)

    if bind.InputType
        ==
        Enum.UserInputType.Keyboard
    then

        return bind.KeyCode.Name

    end


    if bind.InputType
        ==
        Enum.UserInputType.MouseButton1
    then

        return "MOUSE 1"

    end


    if bind.InputType
        ==
        Enum.UserInputType.MouseButton2
    then

        return "MOUSE 2"

    end


    if bind.InputType
        ==
        Enum.UserInputType.MouseButton3
    then

        return "MOUSE 3"

    end


    return bind.InputType.Name

end


local function validBindInput(input)

    if input.UserInputType
        ==
        Enum.UserInputType.Keyboard
    then

        return
            input.KeyCode
            ~=
            Enum.KeyCode.Unknown

    end


    return

        input.UserInputType
        ==
        Enum.UserInputType.MouseButton1

        or

        input.UserInputType
        ==
        Enum.UserInputType.MouseButton2

        or

        input.UserInputType
        ==
        Enum.UserInputType.MouseButton3

end


local function inputToBind(input)

    if input.UserInputType
        ==
        Enum.UserInputType.Keyboard
    then

        return {

            InputType =
                Enum.UserInputType.Keyboard,

            KeyCode =
                input.KeyCode
        }

    end


    return {

        InputType =
            input.UserInputType,

        KeyCode =
            nil
    }

end


local function matchesBind(
    input,
    bind
)

    if bind.InputType
        ==
        Enum.UserInputType.Keyboard
    then

        return

            input.UserInputType
            ==
            Enum.UserInputType.Keyboard

            and

            input.KeyCode
            ==
            bind.KeyCode

    end


    return

        input.UserInputType
        ==
        bind.InputType

end

local function newLine()

    local line =
        Instance.new(
            "Frame"
        )

    line.AnchorPoint =
        Vector2.new(
            0.5,
            0.5
        )

    line.BackgroundColor3 =
        ESP_COLOR

    line.BorderSizePixel = 0

    line.Visible = false

    line.ZIndex = 20

    line.Parent =
        Overlay

    return line

local function setLine(
    line,
    a,
    b,
    thickness
)

    local delta =
        b - a

    if delta.Magnitude <= 0 then

        line.Visible =
            false

        return

    end


    line.Size =
        UDim2.fromOffset(
            delta.Magnitude,
            thickness
        )


    line.Position =
        UDim2.fromOffset(
            (
                a.X
                +
                b.X
            )
            /
            2,

            (
                a.Y
                +
                b.Y
            )
            /
            2
        )


    line.Rotation =
        math.deg(
            math.atan2(
                delta.Y,
                delta.X
            )
        )


    line.BackgroundColor3 =
        ESP_COLOR

    line.Visible =
        true

end


local function hideLines(lines)

    for _, line in ipairs(lines) do

        line.Visible =
            false

    end

end

local function updateBox(
    data,
    character
)

    if not ESP_BOX then

        hideLines(
            data.Box
        )

        return

    end


    local root =
        character:
        FindFirstChild(
            "HumanoidRootPart"
        )

    local head =
        character:
        FindFirstChild(
            "Head"
        )


    if not root
        or not head
    then

        hideLines(
            data.Box
        )

        return

    end


    local rootScreen,
          rootVisible =

        Camera:
        WorldToViewportPoint(
            root.Position
        )


    local topScreen,
          topVisible =

        Camera:
        WorldToViewportPoint(
            head.Position
            +
            Vector3.new(
                0,
                0.8,
                0
            )
        )


    local bottomScreen,
          bottomVisible =

        Camera:
        WorldToViewportPoint(
            root.Position
            -
            Vector3.new(
                0,
                3,
                0
            )
        )


    if not rootVisible
        or not topVisible
        or not bottomVisible
    then

        hideLines(
            data.Box
        )

        return

    end


    local height =
        math.abs(
            bottomScreen.Y
            -
            topScreen.Y
        )


    if height < 5 then

        hideLines(
            data.Box
        )

        return

    end


    local width =
        height * 0.55


    local x =
        rootScreen.X


    local TL =
        Vector2.new(
            x - width / 2,
            topScreen.Y
        )

    local TR =
        Vector2.new(
            x + width / 2,
            topScreen.Y
        )

    local BL =
        Vector2.new(
            x - width / 2,
            bottomScreen.Y
        )

    local BR =
        Vector2.new(
            x + width / 2,
            bottomScreen.Y
        )


    setLine(
        data.Box[1],
        TL,
        TR,
        2
    )

    setLine(
        data.Box[2],
        TR,
        BR,
        2
    )

    setLine(
        data.Box[3],
        BR,
        BL,
        2
    )

    setLine(
        data.Box[4],
        BL,
        TL,
        2
    )

end

local function updateSkeleton(
    data,
    character,
    humanoid
)

    if not ESP_SKELETON then

        hideLines(
            data.Skeleton
        )

        return

    end


    hideLines(
        data.Skeleton
    )


    local definition =

        humanoid.RigType
        ==
        Enum.HumanoidRigType.R15

        and R15
        or R6


    for index, pair in ipairs(
        definition
    ) do

        local a =
            character:
            FindFirstChild(
                pair[1]
            )

        local b =
            character:
            FindFirstChild(
                pair[2]
            )

        local line =
            data.Skeleton[index]


        if a
            and b
            and line
        then

            local sa,
                  va =

                Camera:
                WorldToViewportPoint(
                    a.Position
                )


            local sb,
                  vb =

                Camera:
                WorldToViewportPoint(
                    b.Position
                )


            if va
                and vb
                and sa.Z > 0
                and sb.Z > 0
            then

                setLine(
                    line,

                    Vector2.new(
                        sa.X,
                        sa.Y
                    ),

                    Vector2.new(
                        sb.X,
                        sb.Y
                    ),

                    2
                )

            end

        end

    end

end

    end


    return

local function updateHitbox(player)

    if player == LP
        or isWhitelisted(player)
    then

        removeHitbox(
            player
        )

        return

    end


    if not HITBOX_ENABLED then

        removeHitbox(
            player
        )

        return

    end


    local character =
        player.Character


    if not character then

        removeHitbox(
            player
        )

        return

    end


    local humanoid =
        character:
        FindFirstChildOfClass(
            "Humanoid"
        )


    local target =
        getHitboxTarget(
            character
        )


    if not humanoid
        or humanoid.Health <= 0
        or not target
    then

        removeHitbox(
            player
        )

        return

    end


    local hitbox =
        Hitboxes[player]


    if hitbox then

        local targetValue =
            hitbox:
            FindFirstChild(
                "TargetPart"
            )

        if not targetValue
            or targetValue.Value
            ~= target
        then

            removeHitbox(
                player
            )

            hitbox =
                nil

        end

    end


    if not hitbox then

        hitbox =
            Instance.new(
                "Part"
            )

        hitbox.Name =
            "ZalanDebugHitbox"

        hitbox.Anchored =
            false

        hitbox.CanCollide =
            false

        hitbox.CanTouch =
            false

        hitbox.CanQuery =
            false

        hitbox.Massless =
            true

        hitbox.CastShadow =
            false

        hitbox.Material =
            Enum.Material.ForceField

        hitbox.CFrame =
            target.CFrame

        hitbox.Parent =
            character


        local targetValue =
            Instance.new(
                "ObjectValue"
            )

        targetValue.Name =
            "TargetPart"

        targetValue.Value =
            target

        targetValue.Parent =
            hitbox


        local weld =
            Instance.new(
                "WeldConstraint"
            )

        weld.Part0 =
            target

        weld.Part1 =
            hitbox

        weld.Parent =
            hitbox


        Hitboxes[player] =
            hitbox

    end


    hitbox.Size =
        target.Size
        *
        HITBOX_SCALE


    hitbox.Color =
        ESP_COLOR


    hitbox.Transparency =

        HITBOX_VISIBLE

        and 0.65

        or 1

end

    end

end

--==============================================================

--==============================================================

local function validAimPlayer(
    player
)

    if not player
        or player == LP
    then

        return false

    end


    if isWhitelisted(
        player
    ) then

        return false

    end


    local character =
        player.Character


    if not character then

        return false

    end


    local humanoid =
        character:
        FindFirstChildOfClass(
            "Humanoid"
        )


    local part =
        getAimPart(
            character
        )


    if not humanoid
        or humanoid.Health <= 0
        or not part
    then

        return false

    end


    local distance =
        (
            part.Position
            -
            Camera.CFrame.Position
        ).Magnitude


    if distance
        >
        AIM_DISTANCE
    then

        return false

    end


    local screen,
          visible =

        Camera:
        WorldToViewportPoint(
            part.Position
        )


    if not visible
        or screen.Z <= 0
    then

        return false

    end


    local center =
        Vector2.new(
            Camera.ViewportSize.X
            /
            2,

            Camera.ViewportSize.Y
            /
            2
        )


    local screenDistance =
        (
            Vector2.new(
                screen.X,
                screen.Y
            )
            -
            center
        ).Magnitude


    return

        screenDistance
        <=
        AIM_FOV

end


--==============================================================
-- FIND TARGET
--==============================================================

local function findAimPlayer()

    local closest =
        nil

    local closestDistance =
        math.huge


    local center =
        Vector2.new(
            Camera.ViewportSize.X
            /
            2,

            Camera.ViewportSize.Y
            /
            2
        )


    for _, player in ipairs(
        Players:GetPlayers()
    ) do

        if validAimPlayer(
            player
        ) then

            local part =
                getAimPart(
                    player.Character
                )


            local screen =
                Camera:
                WorldToViewportPoint(
                    part.Position
                )


            local distance =
                (
                    Vector2.new(
                        screen.X,
                        screen.Y
                    )
                    -
                    center
                ).Magnitude


            if distance
                <
                closestDistance
            then

                closestDistance =
                    distance

                closest =
                    player

            end

        end

    end


    return closest

end

local PlayerRows = {}


local function createWhitelistRow(
    player
)

    if player == LP then
        return
    end


    if PlayerRows[player] then

        PlayerRows[player]:
            Destroy()

        PlayerRows[player] =
            nil

    end


    local card =
        createCard(
            WhitelistPage,
            68
        )


    local name =
        makeText(
            card,
            player.DisplayName,
            13,
            TEXT,
            Enum.Font.GothamBold
        )

    name.Size =
        UDim2.new(
            1,
            -190,
            0,
            25
        )

    name.Position =
        UDim2.fromOffset(
            18,
            10
        )


    local username =
        makeText(
            card,
            "@"
            ..
            player.Name,
            10,
            MUTED,
            Enum.Font.GothamMedium
        )

    username.Size =
        UDim2.new(
            1,
            -190,
            0,
            20
        )

    username.Position =
        UDim2.fromOffset(
            18,
            36
        )


    local button =
        Instance.new(
            "TextButton"
        )

    button.Size =
        UDim2.fromOffset(
            145,
            36
        )

    button.Position =
        UDim2.new(
            1,
            -163,
            0.5,
            -18
        )

    button.BorderSizePixel =
        0

    button.AutoButtonColor =
        false

    button.TextSize =
        10

    button.Font =
        Enum.Font.GothamBold

    button.Parent =
        card

    corner(
        button,
        8
    )


    local function refresh()

        if isWhitelisted(
            player
        ) then

            button.Text =
                "WHITELISTED"

            button.BackgroundColor3 =
                GREEN

            button.TextColor3 =
                TEXT

        else

            button.Text =
                "WHITELIST"

            button.BackgroundColor3 =
                CARD2

            button.TextColor3 =
                ACCENT

        end

    end


    connect(
        button.MouseButton1Click,
        function()

            setWhitelisted(
                player,
                not isWhitelisted(
                    player
                )
            )

            refresh()

        end
    )


    refresh()


    PlayerRows[player] =
        card

end


local function refreshWhitelist()

    for player, row in pairs(
        PlayerRows
    ) do

        if not player.Parent then

            row:Destroy()

            PlayerRows[player] =
                nil

        end

    end


    for _, player in ipairs(
        Players:GetPlayers()
    ) do

        if player ~= LP
            and
            not PlayerRows[player]
        then

            createWhitelistRow(
                player
            )

        end

    end

end

--==============================================================
-- ESP + HITBOX RENDER
--==============================================================

connect(
    RunService.RenderStepped,
    function()

        if UNLOADED then
            return
        end


        local myCharacter =
            LP.Character


        local myRoot =
            myCharacter
            and
            myCharacter:
            FindFirstChild(
                "HumanoidRootPart"
            )


        for _, player in ipairs(
            Players:GetPlayers()
        ) do

            if player == LP then
                continue
            end


            if not ESP[player] then

                createESP(
                    player
                )

            end


            local data =
                ESP[player]


            --==================================================
            -- WHITELIST
            --==================================================

            if isWhitelisted(
                player
            ) then

                hideESP(
                    data
                )

                removeHitbox(
                    player
                )

                continue

            end


            --==================================================
            -- DEBUG HITBOX
            --==================================================

            if HITBOX_ENABLED then

                updateHitbox(
                    player
                )

            else

                removeHitbox(
                    player
                )

            end


            --==================================================
            -- ESP OFF
            --==================================================

            if not ESP_ENABLED
                or not myRoot
            then

                hideESP(
                    data
                )

                continue

            end


            local character =
                player.Character


            local humanoid =
                character
                and
                character:
                FindFirstChildOfClass(
                    "Humanoid"
                )


            local root =
                character
                and
                character:
                FindFirstChild(
                    "HumanoidRootPart"
                )


            local head =
                character
                and
                character:
                FindFirstChild(
                    "Head"
                )


            if not character
                or not humanoid
                or not root
                or not head
                or humanoid.Health <= 0
            then

                hideESP(
                    data
                )

                continue

            end


            local distance =
                (
                    myRoot.Position
                    -
                    root.Position
                ).Magnitude


            if distance
                >
                ESP_MAX_DISTANCE
            then

                hideESP(
                    data
                )

                continue

            end


            local screen,
                  visible =

                Camera:
                WorldToViewportPoint(
                    head.Position
                    +
                    Vector3.new(
                        0,
                        0.8,
                        0
                    )
                )


            if not visible
                or screen.Z <= 0
            then

                hideESP(
                    data
                )

                continue

            end


            --==================================================
            -- HIGHLIGHT
            --==================================================

            data.Highlight.Adornee =
                character

            data.Highlight.FillColor =
                ESP_COLOR

            data.Highlight.OutlineColor =
                ESP_COLOR

            data.Highlight.Enabled =
                ESP_HIGHLIGHT


            --==================================================
            -- INFO TEXT
            --==================================================

            local info = {}


            if ESP_NAME then

                table.insert(
                    info,
                    player.DisplayName
                )

            end


            if ESP_DISTANCE then

                table.insert(
                    info,
                    "Távolság: "
                    ..
                    math.floor(
                        distance
                    )
                )

            end


            if ESP_HEALTH then

                table.insert(
                    info,
                    "HP: "
                    ..
                    math.floor(
                        humanoid.Health
                    )
                    ..
                    "/"
                    ..
                    math.floor(
                        humanoid.MaxHealth
                    )
                )

            end


            data.Info.Text =
                table.concat(
                    info,
                    "\n"
                )


            data.Info.TextColor3 =
                ESP_COLOR


            data.Info.Position =
                UDim2.fromOffset(
                    screen.X,
                    screen.Y
                )


            data.Info.Visible =
                #info > 0


            updateBox(
                data,
                character
            )


            updateSkeleton(
                data,
                character,
                humanoid
            )

        end

    end
)


--==============================================================
-- AIM CAMERA LOOP
--==============================================================

RunService:
    BindToRenderStep(

        AIM_RENDER_NAME,

        Enum.RenderPriority.Camera.Value
        +
        1,

        function()

            if UNLOADED then
                return
            end


            if not AIM_ENABLED then

                currentTarget =
                    nil

                return

            end


            if AIM_STICKY then

                if not validAimPlayer(
                    currentTarget
                ) then

                    currentTarget =
                        findAimPlayer()

                end

            else

                currentTarget =
                    findAimPlayer()

            end


            if not validAimPlayer(
                currentTarget
            ) then

                currentTarget =
                    nil

                return

            end


            local targetPart =
                getAimPart(
                    currentTarget.Character
                )


            if not targetPart then
                return
            end


            local wanted =
                CFrame.lookAt(
                    Camera.CFrame.Position,
                    targetPart.Position
                )


            local strength =
                math.clamp(
                    AIM_STRENGTH
                    /
                    100,
                    0.01,
                    1
                )


            Camera.CFrame =
                Camera.CFrame:
                Lerp(
                    wanted,
                    strength
                )

        end
    )


--==============================================================
-- INPUT BEGAN
--==============================================================

connect(
    UIS.InputBegan,
    function(
        input,
        processed
    )

        if UNLOADED then
            return
        end


        --======================================================
        -- KEYBIND CAPTURE
        --======================================================

        if waitingBind then

            if validBindInput(
                input
            ) then

                local callback =
                    waitingBind

                waitingBind =
                    nil

                callback(
                    input
                )

            end

            return

        end


        if processed then
            return
        end


        --======================================================
        -- INSERT
        --======================================================

        if input.UserInputType
            ==
            Enum.UserInputType.Keyboard

            and

            input.KeyCode
            ==
            GUI_KEY
        then

            Main.Visible =
                not Main.Visible

            return

        end


        --======================================================
        -- AIM
        --======================================================

        if matchesBind(
            input,
            AIM_BIND
        ) then

            if AIM_MODE
                ==
                "HOLD"
            then

                AIM_ENABLED =
                    true

            else

                AIM_ENABLED =
                    not AIM_ENABLED


                if not AIM_ENABLED then

                    currentTarget =
                        nil

                end

            end


            return

        end


        --======================================================
        -- SPEED
        --======================================================

        if matchesBind(
            input,
            SPEED_BIND
        ) then

            SPEED_ENABLED =
                not SPEED_ENABLED


            SpeedToggle.Set(
                SPEED_ENABLED
            )


            return

        end

    end
)


--==============================================================
-- INPUT ENDED
--==============================================================

connect(
    UIS.InputEnded,
    function(input)

        if UNLOADED then
            return
        end


        if AIM_MODE
            ==
            "HOLD"

            and

            matchesBind(
                input,
                AIM_BIND
            )
        then

            AIM_ENABLED =
                false

            currentTarget =
                nil

        end

    end
)
